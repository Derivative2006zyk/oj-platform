```python
class Submission(Base):
    __tablename__ = "submissions"
    id: int
    user_id: int (FK users, index)
    problem_id: int (FK problems, index)
    code: str
    language: str
    status: str (index)  # PENDING/AC/WA/TLE/MLE/RE/CE
    runtime_ms: Optional[int]
    memory_kb: Optional[int]
    passed_cases: int
    total_cases: int
    error_message: Optional[str]
    created_at: datetime
```
